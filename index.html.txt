  <!-- EDITOR PANEL -->
  <div id="editorView">

<script>
  const defaultStops = [
    {
      title: "Zhangjiajie Pillars",
      mission: "Explore the sandstone peak forest in Hunan Province. These towering pillars look like natural skyscrapers!",
      linkUrl: "https://en.wikipedia.org/wiki/Zhangjiajie_National_Forest_Park",
      linkText: "Zhangjiajie Geology Guide",
      hint: "Search for or visit the link above to learn how quartz sandstone pillars were formed.",
      question: "What process primarily shaped these tall quartz sandstone pillars over millions of years?",
      answers: ["erosion", "water erosion", "weathering"],
      badgeIcon: "🏛️",
      badgeTitle: "Sandstone Scout"
    },
    {
      title: "Zhangye Danxia",
      mission: "Travel to Gansu Province to examine the vivid, multicolored rock layers of the Danxia landforms.",
      linkUrl: "https://en.wikipedia.org/wiki/Zhangye_National_Geopark",
      linkText: "Zhangye Danxia Landforms",
      hint: "Check the resource to find out what gives the rocks their striking red color.",
      question: "Which mineral/element oxidation gives these sandstone layers their red pigment?",
      answers: ["iron", "iron oxide", "rust"],
      badgeIcon: "🌈",
      badgeTitle: "Mineral Analyst"
    },
    {
      title: "Guilin Karst",
      mission: "Navigate the Li River in Guangxi to analyze the distinct, rounded Karst mountains and underground caves.",
      linkUrl: "https://en.wikipedia.org/wiki/Guilin",
      linkText: "Guilin Karst Formation",
      hint: "Look up what type of soluble rock forms karst caves and tower peaks.",
      question: "What soft rock type dissolves in rainwater to create karst caves and peaks?",
      answers: ["limestone", "calcite"],
      badgeIcon: "🦇",
      badgeTitle: "Karst Caver"
    }
  ];

  // Safe localStorage helper to prevent local file security errors
  function getStoredData() {
    try {
      const stored = localStorage.getItem('webquest_data');
      return stored ? JSON.parse(stored) : null;
    } catch (e) {
      console.warn("LocalStorage access restricted on local file system.");
      return null;
    }
  }

  function setStoredData(data) {
    try {
      localStorage.setItem('webquest_data', JSON.stringify(data));
    } catch (e) {
      console.warn("Could not save to localStorage on local file system.");
    }
  }

  let questData = getStoredData() || {
    title: "🌏 China Geology WebQuest",
    subtitle: "Mission Briefing: Investigate China's iconic landscapes and collect field badges!",
    stops: defaultStops
  };

  let completedStops = [];
  let isEditing = false;

  function init() {
    document.getElementById('questTitle').textContent = questData.title;
    document.getElementById('questSubtitle').textContent = questData.subtitle;

    if (completedStops.length !== questData.stops.length) {
      completedStops = new Array(questData.stops.length).fill(false);
    }

    renderTabs();
    renderStops();
    renderBadges();
    switchTab(0);
  }

  function renderTabs() {
    const tabContainer = document.getElementById('tabContainer');
    tabContainer.innerHTML = '';
    
    questData.stops.forEach((stop, index) => {
      const btn = document.createElement('button');
      btn.className = `tab-btn ${index === 0 ? 'active' : ''}`;
      btn.textContent = `Stop ${index + 1}: ${stop.title}`;
      btn.onclick = () => switchTab(index);
      tabContainer.appendChild(btn);
    });

    const logBtn = document.createElement('button');
    logBtn.className = 'tab-btn';
    logBtn.textContent = 'Mission Log';
    logBtn.onclick = () => switchTab(questData.stops.length);
    tabContainer.appendChild(logBtn);
  }

  function renderStops() {
    const container = document.getElementById('stopContentContainer');
    container.innerHTML = '';

    questData.stops.forEach((stop, index) => {
      const stopDiv = document.createElement('div');
      stopDiv.id = `stop-${index}`;
      stopDiv.style.display = index === 0 ? 'block' : 'none';

      const linkHtml = stop.linkUrl 
        ? `<p>🔗 <strong>Field Resource:</strong> <a href="${stop.linkUrl}" target="_blank" rel="noopener">${stop.linkText || stop.linkUrl}</a></p>`
        : '';

      stopDiv.innerHTML = `
        <h2>Stop ${index + 1}: ${stop.title}</h2>
        <div class="mission-card">
          <p><strong>Your Mission:</strong> ${stop.mission}</p>
          ${linkHtml}
          <div class="hint-box">🔍 <strong>Field Hint:</strong> ${stop.hint}</div>
          <label><strong>Question:</strong> ${stop.question}</label>
          <input type="text" id="input-${index}" placeholder="Type your answer here...">
          <button class="btn-action" onclick="checkAnswer(${index})">Submit Discovery</button>
          <div id="feedback-${index}" class="feedback"></div>
        </div>
      `;
      container.appendChild(stopDiv);
    });
  }

  function renderBadges() {
    const badgeContainer = document.getElementById('badgeContainer');
    badgeContainer.innerHTML = '';

    questData.stops.forEach((stop, index) => {
      const badgeDiv = document.createElement('div');
      badgeDiv.className = `badge ${completedStops[index] ? 'unlocked' : ''}`;
      badgeDiv.id = `badge-${index}`;
      badgeDiv.innerHTML = `
        <div class="badge-icon">${stop.badgeIcon || '⭐'}</div>
        <small>${stop.badgeTitle || 'Badge ' + (index + 1)}</small>
      `;
      badgeContainer.appendChild(badgeDiv);
    });
  }

  function switchTab(index) {
    const tabs = document.querySelectorAll('.tab-btn');
    tabs.forEach((tab, i) => tab.classList.toggle('active', i === index));

    const totalStops = questData.stops.length;
    for (let i = 0; i < totalStops; i++) {
      const el = document.getElementById(`stop-${i}`);
      if (el) el.style.display = (i === index) ? 'block' : 'none';
    }

    const logTab = document.getElementById('missionLogTab');
    logTab.style.display = (index === totalStops) ? 'block' : 'none';
  }

  function checkAnswer(index) {
    const inputVal = document.getElementById(`input-${index}`).value.toLowerCase().trim();
    const feedback = document.getElementById(`feedback-${index}`);
    const validAnswers = questData.stops[index].answers;

    const isCorrect = validAnswers.some(ans => inputVal.includes(ans.toLowerCase().trim()));

    if (isCorrect) {
      feedback.textContent = "✅ Correct! Research field data logged.";
      feedback.className = "feedback correct";
      completedStops[index] = true;
      renderBadges();
      checkCompletion();
    } else {
      feedback.textContent = "❌ Not quite. Review your links/hints and try again!";
      feedback.className = "feedback incorrect";
    }
  }

  function checkCompletion() {
    if (completedStops.every(status => status === true)) {
      document.getElementById('finalScreen').style.display = 'block';
    }
  }

  /* --- EDITOR FUNCTIONS --- */

  function toggleEditor() {
    const playView = document.getElementById('playView');
    const editorView = document.getElementById('editorView');
    const btn = document.getElementById('toggleEditBtn');

    if (isEditing) {
      saveQuest();
    } else {
      isEditing = true;
      btn.textContent = "💾 Save Quest";
      document.getElementById('editTitle').value = questData.title;
      document.getElementById('editSubtitle').value = questData.subtitle;
      renderEditorList();
      playView.style.display = 'none';
      editorView.style.display = 'block';
    }
  }

  function renderEditorList() {
    const list = document.getElementById('editorList');
    list.innerHTML = '';

    questData.stops.forEach((stop, i) => {
      const item = document.createElement('div');
      item.className = 'editor-item';
      item.innerHTML = `
        <h4>Stop ${i + 1} Settings</h4>
        <label>Stop Name:</label>
        <input type="text" id="edit-title-${i}" value="${stop.title}">
        
        <label>Mission Description:</label>
        <textarea id="edit-mission-${i}" rows="2">${stop.mission}</textarea>
        
        <label>Resource Website URL (Optional):</label>
        <input type="text" id="edit-url-${i}" value="${stop.linkUrl || ''}" placeholder="https://...">

        <label>Resource Link Label:</label>
        <input type="text" id="edit-urltext-${i}" value="${stop.linkText || ''}" placeholder="e.g. Read Geology Guide">

        <label>Search / Field Hint:</label>
        <input type="text" id="edit-hint-${i}" value="${stop.hint}">

        <label>Question:</label>
        <input type="text" id="edit-question-${i}" value="${stop.question}">

        <label>Valid Answers (comma-separated):</label>
        <input type="text" id="edit-answers-${i}" value="${stop.answers.join(', ')}">

        <div style="display:flex; gap: 10px;">
          <div style="flex:1;">
            <label>Badge Emoji/Icon:</label>
            <input type="text" id="edit-badgeicon-${i}" value="${stop.badgeIcon || '⭐'}">
          </div>
          <div style="flex:2;">
            <label>Badge Title:</label>
            <input type="text" id="edit-badgetitle-${i}" value="${stop.badgeTitle || ''}">
          </div>
        </div>

        <button class="btn-action btn-danger" style="margin-top:10px;" onclick="removeStop(${i})">Delete Stop</button>
      `;
      list.appendChild(item);
    });
  }

  function syncEditorInputs() {
    questData.stops.forEach((stop, i) => {
      const titleEl = document.getElementById(`edit-title-${i}`);
      if (titleEl) {
        stop.title = titleEl.value;
        stop.mission = document.getElementById(`edit-mission-${i}`).value;
        stop.linkUrl = document.getElementById(`edit-url-${i}`).value;
        stop.linkText = document.getElementById(`edit-urltext-${i}`).value;
        stop.hint = document.getElementById(`edit-hint-${i}`).value;
        stop.question = document.getElementById(`edit-question-${i}`).value;
        stop.answers = document.getElementById(`edit-answers-${i}`).value.split(',').map(s => s.trim());
        stop.badgeIcon = document.getElementById(`edit-badgeicon-${i}`).value;
        stop.badgeTitle = document.getElementById(`edit-badgetitle-${i}`).value;
      }
    });
  }

  function addStop() {
    syncEditorInputs();
    questData.stops.push({
      title: "New Location",
      mission: "Describe what students should explore here.",
      linkUrl: "",
      linkText: "",
      hint: "Add a tip or website guidance.",
      question: "Enter your question here?",
      answers: ["answer1", "answer2"],
      badgeIcon: "🏆",
      badgeTitle: "New Badge"
    });
    renderEditorList();
  }

  function removeStop(index) {
    syncEditorInputs();
    questData.stops.splice(index, 1);
    renderEditorList();
  }

  function saveQuest() {
    syncEditorInputs();
    questData.title = document.getElementById('editTitle').value;
    questData.subtitle = document.getElementById('editSubtitle').value;

    setStoredData(questData);
    completedStops = new Array(questData.stops.length).fill(false);
    
    isEditing = false;
    document.getElementById('toggleEditBtn').textContent = "⚙️ Edit WebQuest";
    document.getElementById('editorView').style.display = 'none';
    document.getElementById('playView').style.display = 'block';
    
    init();
  }

  init();
</script>

</body>
